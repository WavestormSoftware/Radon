add_library(radon_demo_common STATIC common/runner.cpp common/gallery.cpp)
target_include_directories(radon_demo_common PUBLIC common)
target_link_libraries(radon_demo_common PUBLIC radon radon_sdl3_gl)
rd_warnings(radon_demo_common)

add_subdirectory(desktop_demo)
if(CMAKE_SYSTEM_NAME STREQUAL "Emscripten")
    add_subdirectory(web_demo)
endif()
if(ANDROID OR IOS)
    add_subdirectory(mobile_demo)
endif()

# Small standalone examples (desktop only): radon_ex_<name>
if(NOT (CMAKE_SYSTEM_NAME STREQUAL "Emscripten" OR ANDROID OR IOS))
    file(GLOB _rd_examples CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/snippets/*.cpp)
    add_subdirectory(showcase)
    foreach(_src ${_rd_examples})
        get_filename_component(_name ${_src} NAME_WE)
        add_executable(radon_ex_${_name} ${_src})
        target_link_libraries(radon_ex_${_name} PRIVATE radon_demo_common)
        rd_warnings(radon_ex_${_name})
    endforeach()
endif()
